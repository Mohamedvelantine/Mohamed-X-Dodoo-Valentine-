<script>
// زر NO يهرب ويكبر ويختفي
function moveNo(){
  var btn=document.getElementById("no");
  btn.style.top=Math.random()*80+"%";
  btn.style.left=Math.random()*80+"%";
  btn.style.transform="scale("+(0.5+Math.random()*0.8)+")";
  if(Math.random()<0.3){ // أحيانًا يختفي
    btn.style.display="none";
    setTimeout(()=>{btn.style.display="block"}, 2000);
  }
}

// عرض صفحة الحب
function showLove(){
  document.getElementById("page1").style.display="none";
  document.getElementById("lovePage").style.display="flex";
  createHearts(30); // قلوب طائرة
  createConfetti(50); // confetti
}

// إنشاء القلوب الطائرة
function createHearts(num){
  for(let i=0;i<num;i++){
    let heart=document.createElement("div");
    heart.className="heart";
    heart.style.left=Math.random()*window.innerWidth+"px";
    heart.style.animationDuration=(3+Math.random()*3)+"s";
    heart.style.opacity=Math.random();
    document.body.appendChild(heart);
  }
}

// إنشاء confetti
function createConfetti(num){
  for(let i=0;i<num;i++){
    let conf=document.createElement("div");
    conf.className="confetti";
    conf.style.background=["#ff0","#f0f","#0ff","#fff"][Math.floor(Math.random()*4)];
    conf.style.left=Math.random()*window.innerWidth+"px";
    conf.style.animationDuration=(2+Math.random()*2)+"s";
    document.body.appendChild(conf);
  }
}
